GPU Server 도커 이미지 빌드를 위한 도커파일 작성

GPU 서버에 설치될 도커의 이미지를 빌드하기 위해 Docker 파일을 준비했습니다.

2025. 12. 15.
#docker#GPU#Image
# syntax=docker/dockerfile:1
FROM ubuntu:26.04

ENV DEBIAN_FRONTEND=noninteractive

ARG TZ=Asia/Seoul
ARG LOCALE=ko_KR.UTF-8

RUN set -eux; \
  apt-get update; \
  apt-get install -y --no-install-recommends \
    ca-certificates curl wget git vim \
    tzdata locales \
    build-essential pkg-config \
    net-tools iputils-ping \
    sudo \
    openssh-server \
    # Web stack (install only)
    nginx apache2 \
    php php-cli php-fpm php-common \
    # DB (install only)
    postgresql postgresql-contrib \
    # (optional) python/node
    python3 python3-pip python-is-python3 \
    nodejs npm \
  ; \
  ln -snf /usr/share/zoneinfo/${TZ} /etc/localtime; \
  echo "${TZ}" > /etc/timezone; \
  sed -i "s/^# *\(ko_KR\.UTF-8 UTF-8\)/\1/" /etc/locale.gen; \
  locale-gen; \
  apt-get clean; \
  rm -rf /var/lib/apt/lists/* /tmp/* /var/tmp/*

ENV LANG=ko_KR.UTF-8 \
    LC_ALL=ko_KR.UTF-8

# SSH 최소 준비(설정 “과다” 없이, 접속만 되게)
RUN set -eux; \
  mkdir -p /run/sshd

# dev 유저(SSH 로그인용)
ARG USER=dev
ARG UID=1000
ARG GID=1000
ARG DEV_PASSWORD=dev
RUN set -eux; \
  groupadd -g ${GID} ${USER}; \
  useradd -m -u ${UID} -g ${GID} -s /bin/bash ${USER}; \
  usermod -aG sudo ${USER}; \
  echo "${USER}:${DEV_PASSWORD}" | chpasswd

EXPOSE 22
CMD ["/usr/sbin/sshd", "-D"]
